Array and Function: Write a function that takes an array of numbers and returns the sum of all elements.
Ans:
function sumArray(numbers) {
    let sum = 0;

    for (let i = 0; i < numbers.length; i++) {
        sum = sum + numbers[i];
    }

    return sum;
}

console.log(sumArray([10, 20, 30, 40, 50]));
